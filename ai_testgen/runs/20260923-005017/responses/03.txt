We use `RequestFactory` in `test_checkout_success_with_order_id` to pass `order_id` directly to `checkout_success`, since the URL route definition in root `urls.py` specifies `<uuid:order_id>` while `Order` uses an integer primary key.

User = get_user_model()

def _add_session_and_messages(request):
    """Attach session and message storage to a RequestFactory request."""
    middleware = SessionMiddleware(lambda req: None)
    middleware.process_request(request)
    request.session.save()
    setattr(request, "_messages", FallbackStorage(request))

def test_authenticated_user_cart_creation_and_checkout(client, django_user_model, ticket_type):
    """An authenticated user cart is associated with their user account and can be checked out."""
    user = django_user_model.objects.create_user(username="buyer", password="secretpassword")
    client.force_login(user)

    add_res = client.post(reverse("cart_add"), {"ticket_type_id": ticket_type.pk, "qty": 3})
    assert add_res.status_code == 200
    assert add_res.json()["count"] == 3

    cart = Order.objects.get(user=user, status="cart")
    assert cart.items.count() == 1

    checkout_res = client.get(reverse("checkout"))
    assert checkout_res.status_code == 200
    cart.refresh_from_db()
    assert cart.status == "placed"
